# Quick Reference - Production-Ready PHP Auth

## Setup
cp .env.example .env           # Create local config
nano .env                       # Add your credentials
bash setup.sh                   # Initialize everything

## Development
export APP_ENV=development      # Enable verbose logging
php -S localhost:8000          # Start server
tail -f log.txt               # Watch logs (will be verbose)

## Testing  
php tests/run-tests.php        # Run unit tests
bash tests/api-tests.sh       # Run endpoint tests

## Production
export APP_ENV=production       # Disable debug logging
php -S 0.0.0.0:8000           # Start on all interfaces
tail -f log.txt               # Watch logs (minimal)

## Security Check
git status                      # Verify no .env or config.php are shown
git log -1                     # Check last commit has no secrets

## Push to GitHub
git add .
git commit -m "Your message"
git push origin main

## Environments
# development  = Verbose logging, see everything
# staging      = Minimal logging, errors only
# production   = Secure, no debug output
